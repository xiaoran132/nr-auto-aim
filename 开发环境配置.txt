安装Cmake4.2，防止出现不兼容问题
cd ~/Cmake
wget https://github.com/Kitware/CMake/releases/download/v4.2.5/cmake-4.2.5-linux-x86_64.tar.gz
tar -xzf cmake-4.2.5-linux-x86_64.tar.gz


sudo mv /usr/bin/cmake /usr/bin/cmake.system.bak


# 注册系统自带的 cmake（如果之前没注册过，先确认路径 /usr/bin/cmake 是否指向系统版本）
sudo update-alternatives --install /usr/bin/cmake cmake /usr/bin/cmake 10

# 注册你之前安装的 4.5
sudo update-alternatives --install /usr/bin/cmake cmake /home/xwct/Cmake/cmake-4.5.0-rc1-linux-x86_64/bin/cmake 50

# 注册新下载的 4.2
sudo update-alternatives --install /usr/bin/cmake cmake /home/xwct/Cmake/cmake-4.2.5-linux-x86_64/bin/cmake 40

# 这行命令会精准删除包含 cmake-4.5 的那行 PATH 设置
sed -i '/cmake-4.5/d' ~/.bashrc

# 刷新环境变量
source ~/.bashrc


注册完成后，用以下命令交互式切换：
sudo update-alternatives --config cmake
